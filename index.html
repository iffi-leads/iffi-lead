// IOCH WhatsApp CRM backend (Meta WhatsApp Cloud API) // Setup: npm i express jsonwebtoken axios bcryptjs // Env: WA_TOKEN, VERIFY_TOKEN, JWT_SECRET, SUPER_PASS // Webhook URL in Meta dashboard: https://YOUR-DOMAIN/webhook (subscribe to "messages") const express = require('express'), jwt = require('jsonwebtoken'), axios = require('axios'); const { WA_TOKEN, VERIFY_TOKEN, JWT_SECRET, SUPER_PASS } = process.env; const G = 'https://graph.facebook.com/v21.0'; // use the current Graph API version const app = express(); app.use(express.json()); app.set('trust proxy', true); // Replace with a real database (Postgres/MongoDB). Store hashed passwords (bcrypt). // numbers: {id, label, branch, phone, phoneId (Meta phone_number_id), hung} const db = { users: [], numbers: [], chats: [], log: [] }; const log = (req, a) => db.log.push({ t: new Date(), user: req.user?.name, ip: req.ip, a }); const mask = p => p.slice(0, 4) + '•••••' + p.slice(-2); // ---- Auth ---- app.post('/api/login', (req, res) => { const { username, password } = req.body; let u = null; if (username === 'superadmin' && password === SUPER_PASS) u = { name: 'Super Admin', role: 'S' }; else { const x = db.users.find(x => x.username === username && x.pass === password); // use bcrypt.compare if (x) u = { name: x.name, role: x.role === 'Admin' ? 'A' : 'L', branch: x.branch }; } if (!u) return res.sendStatus(401); req.user = u; log(req, 'login'); res.json({ token: jwt.sign(u, JWT_SECRET, { expiresIn: '12h' }), role: u.role }); }); const auth = (req, res, next) => { try { req.user = jwt.verify((req.headers.authorization || '').slice(7), JWT_SECRET); next(); } catch { res.sendStatus(401); } }; const su = (req, res, next) => req.user.role === 'S' ? next() : res.sendStatus(403); // ---- Meta webhook ---- app.get('/webhook', (q, r) => q.query['hub.verify_token'] === VERIFY_TOKEN ? r.send(q.query['hub.challenge']) : r.sendStatus(403)); app.post('/webhook', (req, res) => { res.sendStatus(200); for (const e of req.body.entry || []) for (const ch of e.changes || []) { const v = ch.value, num = db.numbers.find(n => n.phoneId === v.metadata?.phone_number_id); if (!num) continue; for (const m of v.messages || []) { let chat = db.chats.find(c => c.numId === num.id && c.phone === m.from); if (!chat) db.chats.push(chat = { id: Date.now() + Math.floor(Math.random() * 999), numId: num.id, phone: m.from, name: v.contacts?.[0]?.profile?.name || m.from, msgs: [] }); const msg = { dir: 'in', t: new Date(), type: m.type }; if (m.type === 'text') msg.text = m.text.body; else if (m.type === 'location') { msg.type = 'loc'; msg.lat = m.location.latitude; msg.lng = m.location.longitude; } // only when the customer shares it else if (m.type === 'audio') { msg.type = 'voice'; msg.mediaId = m.audio.id; } chat.msgs.push(msg); } } }); // ---- Chats: role filtering + number masking happen HERE (server side) ---- app.get('/api/chats', auth, (req, res) => { const S = req.user.role === 'S', nums = db.numbers.filter(n => S || n.branch === req.user.branch); res.json({ numbers: nums.map(n => ({ id: n.id, label: n.label, branch: n.branch, hung: n.hung, phone: S ? n.phone : mask(n.phone) })), chats: db.chats.filter(c => nums.some(n => n.id === c.numId)).map(c => ({ ...c, phone: S ? c.phone : mask(c.phone), msgs: c.msgs.map(m => S || m.type !== 'loc' ? m : { ...m, lat: undefined, lng: undefined }) })) }); }); // Play voice notes (staff can listen, nobody can delete) app.get('/api/media/:id', auth, async (req, res) => { const m = await axios.get(`${G}/${req.params.id}`, { headers: { Authorization: 'Bearer ' + WA_TOKEN } }); const f = await axios.get(m.data.url, { headers: { Authorization: 'Bearer ' + WA_TOKEN }, responseType: 'stream' }); res.type(m.data.mime_type); f.data.pipe(res); }); // ---- Send ---- app.post('/api/send', auth, async (req, res) => { const c = db.chats.find(x => x.id == req.body.chat), n = db.numbers.find(x => x.id === c?.numId); if (!n) return res.sendStatus(404); const S = req.user.role === 'S'; if (!S && (n.branch !== req.user.branch || n.hung)) return res.status(403).json({ error: 'Chat paused or not allowed' }); if (req.body.type !== 'text' && !S) return res.sendStatus(403); // only Super Admin sends voice // Free-form replies only work within 24h of the customer's last message; otherwise use an approved template. // Voice: upload ogg/opus audio to POST {G}/{phoneId}/media, then send {type:'audio', audio:{id}} (convert browser webm with ffmpeg). await axios.post(`${G}/${n.phoneId}/messages`, { messaging_product: 'whatsapp', to: c.phone, type: 'text', text: { body: req.body.text } }, { headers: { Authorization: 'Bearer ' + WA_TOKEN } }); c.msgs.push({ dir: 'out', t: new Date(), type: 'text', text: req.body.text, by: req.user.name }); log(req, 'send'); res.sendStatus(200); }); // ---- Super Admin only ---- app.post('/api/hang/:id', auth, su, (q, s) => { const n = db.numbers.find(x => x.id == q.params.id); n.hung = !n.hung; log(q, 'hang=' + n.hung); s.json({ hung: n.hung }); }); app.delete('/api/chats/:id', auth, su, (q, s) => { db.chats = db.chats.filter(c => c.id != q.params.id); log(q, 'delete chat'); s.sendStatus(200); }); app.get('/api/log', auth, su, (q, s) => s.json(db.log)); // includes IP of every staff action app.listen(3000, () => console.log('IOCH CRM backend on :3000'));
